cmake_minimum_required(VERSION 3.24)
project(FileAnalysis VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use MSVC targeting Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /permissive- /utf-8 /EHsc)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_library(analysis STATIC src/core.cpp src/process.cpp src/provenance.cpp src/presentation.cpp src/photo.cpp src/insights.cpp src/previews.cpp src/investigation.cpp src/comparison_view.cpp src/generation_formats.cpp)
target_include_directories(analysis PUBLIC src vendor)
target_link_libraries(analysis PUBLIC bcrypt windowscodecs ole32)
add_executable(file-analysis WIN32 src/gui.cpp src/preview_window.cpp src/resources.rc src/app.manifest)
configure_file(src/version.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/version.h" @ONLY)
target_include_directories(file-analysis PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(file-analysis PRIVATE FA_VERSION="${PROJECT_VERSION}")
target_link_libraries(file-analysis PRIVATE analysis comctl32 comdlg32 shell32 ole32)
target_link_options(file-analysis PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest)
get_filename_component(FA_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
get_filename_component(FA_VC_ROOT "${FA_COMPILER_DIR}/../../../../../.." ABSOLUTE)
file(GLOB FA_VCRUNTIME "${FA_VC_ROOT}/Redist/MSVC/*/x64/Microsoft.VC143.CRT/vcruntime140.dll")
list(SORT FA_VCRUNTIME COMPARE NATURAL ORDER DESCENDING)
if(NOT FA_VCRUNTIME)
 message(FATAL_ERROR "Visual C++ x64 Redistributable files are required for the official c2patool companion")
endif()
list(GET FA_VCRUNTIME 0 FA_VCRUNTIME_DLL)
add_custom_command(TARGET file-analysis POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:file-analysis>/exiftool"
 COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/tools/exiftool/exiftool.exe" "$<TARGET_FILE_DIR:file-analysis>/exiftool/exiftool.exe"
 COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/tools/exiftool/exiftool_files" "$<TARGET_FILE_DIR:file-analysis>/exiftool/exiftool_files"
 COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:file-analysis>/c2patool")
add_custom_command(TARGET file-analysis POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FA_VCRUNTIME_DLL}" "$<TARGET_FILE_DIR:file-analysis>/c2patool/vcruntime140.dll")
foreach(asset c2patool.exe offline.toml c2pa-trust-list.pem LICENSE-MIT LICENSE-APACHE)
 add_custom_command(TARGET file-analysis POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/tools/c2patool/${asset}" "$<TARGET_FILE_DIR:file-analysis>/c2patool/${asset}")
endforeach()
add_executable(analysis-tests tests/tests.cpp)
target_link_libraries(analysis-tests PRIVATE analysis)
add_executable(provenance-tests tests/provenance_tests.cpp)
target_link_libraries(provenance-tests PRIVATE analysis)
add_executable(presentation-tests tests/presentation_tests.cpp)
target_link_libraries(presentation-tests PRIVATE analysis)
add_executable(photo-tests tests/photo_tests.cpp)
target_link_libraries(photo-tests PRIVATE analysis)
add_executable(insights-tests tests/insights_tests.cpp)
target_link_libraries(insights-tests PRIVATE analysis)
add_executable(preview-tests tests/preview_tests.cpp)
target_link_libraries(preview-tests PRIVATE analysis)
add_executable(investigation-tests tests/investigation_tests.cpp)
target_link_libraries(investigation-tests PRIVATE analysis)
add_executable(comparison-tests tests/comparison_tests.cpp)
target_link_libraries(comparison-tests PRIVATE analysis)
add_executable(generation-tests tests/generation_formats_tests.cpp)
target_link_libraries(generation-tests PRIVATE analysis)
add_executable(format-tests tests/format_tests.cpp)
target_link_libraries(format-tests PRIVATE analysis)
add_executable(thread-sample EXCLUDE_FROM_ALL tests/thread_sample.cpp)
enable_testing()
add_test(NAME engine COMMAND analysis-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures" "$<TARGET_FILE_DIR:file-analysis>/exiftool/exiftool.exe")
add_test(NAME provenance COMMAND provenance-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures" "$<TARGET_FILE_DIR:file-analysis>/exiftool/exiftool.exe" "$<TARGET_FILE_DIR:file-analysis>/c2patool/c2patool.exe")
add_test(NAME presentation COMMAND presentation-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
add_test(NAME photo COMMAND photo-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures" "$<TARGET_FILE_DIR:file-analysis>/exiftool/exiftool.exe")
add_test(NAME insights COMMAND insights-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures" "$<TARGET_FILE_DIR:file-analysis>/exiftool/exiftool.exe")
add_test(NAME previews COMMAND preview-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures" "$<TARGET_FILE_DIR:file-analysis>/exiftool/exiftool.exe")
add_test(NAME investigation COMMAND investigation-tests)
add_test(NAME comparison COMMAND comparison-tests)
add_test(NAME generation COMMAND generation-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
add_test(NAME formats COMMAND format-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures" "$<TARGET_FILE_DIR:file-analysis>/exiftool/exiftool.exe")
install(TARGETS file-analysis RUNTIME DESTINATION .)
install(DIRECTORY tools/exiftool/exiftool_files DESTINATION exiftool)
install(FILES tools/exiftool/exiftool.exe tools/exiftool/README.txt DESTINATION exiftool)
install(FILES tools/c2patool/c2patool.exe tools/c2patool/offline.toml tools/c2patool/c2pa-trust-list.pem tools/c2patool/LICENSE-MIT tools/c2patool/LICENSE-APACHE DESTINATION c2patool)
install(FILES "${FA_VCRUNTIME_DLL}" DESTINATION c2patool)
install(FILES README.md THIRD_PARTY.md DEPENDENCIES.json vendor/JSON-LICENSE.txt DESTINATION .)
install(FILES docs/RESEARCH.md docs/GUI-TESTING.md docs/RELEASE-${PROJECT_VERSION}.md DESTINATION docs)
install(FILES docs/FORMAT-COVERAGE.md docs/AUDIT-1.0.md DESTINATION docs)
